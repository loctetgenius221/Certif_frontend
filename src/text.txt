import { ref, onMounted } from 'vue';
import { useRoute } from 'vue-router';
import Swal from 'sweetalert2';
import SlideModal from "@/components/SlideModal.vue";

import HeaderPatient from "@/components/HeaderPatient.vue";
import {
  getDossierMedical,

} from "@/services/dossiermedicalService";

// État initial
const documents = ref([]);
const patient = ref({
  user: null,
  numero_patient: '',
  numero_urgence: ''
});
const dossier = ref(null);
const isLoading = ref(true);
const error = ref(null);
const route = useRoute();
const file = ref(null);
const type_document = ref("");
const dossier_medical_id = ref(null);
const isFileValid = ref(false);

// État du modal
const isModalOpen = ref(false);
const modalType = ref('');
const modalTitle = ref('');

// Formatage des dates
const formatDate = (dateString) => {
  if (!dateString) return 'Non spécifiée';
  return new Date(dateString).toLocaleDateString('fr-FR');
};

// Récupération du dossier médical
const fetchDossierMedical = async () => {
  isLoading.value = true;
  error.value = null;
  const dossierId = route.params.id;
  dossier_medical_id.value = dossierId;

  try {
    const response = await getDossierMedical(dossierId);
    dossier.value = response.data;
    documents.value = response.data.documents || [];
    patient.value = response.data.patient || { user: null };
  } catch (err) {
    console.error("Erreur lors de la récupération du dossier médical:", err);
    error.value = "Impossible de charger le dossier médical. Veuillez réessayer.";
  } finally {
    isLoading.value = false;
  }
};

// Gestion des fichiers
const onFileChange = (event) => {
  file.value = event.target.files[0];
  isFileValid.value = !!file.value;
};

// Upload de fichiers
const handleFileUpload = async () => {
  try {
    if (!file.value || !type_document.value) {
      Swal.fire({
        icon: 'error',
        title: 'Erreur',
        text: 'Veuillez sélectionner un fichier et renseigner le type de document.'
      });
      return;
    }

    const response = await uploadDocumentByMedecin(
      dossier_medical_id.value,
      type_document.value,
      file.value
    );

    Swal.fire({
      icon: 'success',
      title: 'Succès',
      text: 'Document téléversé avec succès !'
    });

    // Réinitialisation des champs
    file.value = null;
    type_document.value = '';
    isFileValid.value = false;
    
    // Rafraîchir la liste des documents
    await fetchDossierMedical();
  } catch (error) {
    console.error("Erreur lors de l'ajout du document:", error);
    Swal.fire({
      icon: 'error',
      title: 'Erreur',
      text: 'Une erreur est survenue lors du téléversement du document.'
    });
  }
};

// Gestion du modal
const openModal = (type) => {
  modalType.value = type;
  switch(type) {
    case 'antecedents':
      modalTitle.value = 'Ajouter un antécédent médical';
      break;
    case 'traitements':
      modalTitle.value = 'Ajouter un traitement';
      break;
    case 'notes':
      modalTitle.value = 'Ajouter une note d\'observation';
      break;
    case 'interventions':
      modalTitle.value = 'Ajouter une intervention chirurgicale';
      break;
    case 'infos':
      modalTitle.value = 'Ajouter une information supplémentaire';
      break;
  }
  isModalOpen.value = true;
};

const closeModal = () => {
  isModalOpen.value = false;
};

  try {
    switch(modalType.value) {
      case 'antecedents':

          dossier.value.antecedents_medicaux = [];
        }
        dossier.value.antecedents_medicaux.push(formData.antecedent);
        break;
      case 'traitements':
        if (!dossier.value.traitements) {
            case 'traitements':

      break;
    case 'infos':
      modalTitle.value = 'Ajouter une information supplémentaire';
      break;
    default:
      modalTitle.value = '';
  }
  isModalOpen.value = true;
};

const closeModal = () => {
